cmake_minimum_required(VERSION 3.16)
project(geist_whisper_candidate LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
if(NOT WHISPER_SOURCE)
  message(FATAL_ERROR "Use scripts/build-whisper.sh to supply the pinned whisper.cpp source")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
add_subdirectory("${WHISPER_SOURCE}" whisper-engine)
add_executable(diktat-whisper ../src/whisper_diktat.cpp)
target_compile_options(diktat-whisper PRIVATE -Wall -Wextra)
target_link_libraries(diktat-whisper PRIVATE whisper Threads::Threads)
